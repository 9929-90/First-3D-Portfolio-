    98→    98→function useScramble(text, isHovered) {
    99→    99→  const [display, setDisplay] = useState(text);
   100→   100→  const frameRef = useRef(null);
   101→   101→  const iterRef = useRef(0);
   102→   102→
   103→   103→  useEffect(() => {
   104→   104→    if (!isHovered) { setDisplay(text); return; }
   105→   105→    iterRef.current = 0;
   106→   106→    const scramble = () => {
   107→   107→      const iter = iterRef.current;
   108→   108→      setDisplay(
   109→   109→        text.split('').map((char, i) => {
   110→   110→          if (char === ' ') return ' ';
   111→   111→          if (i < iter) return text[i];
   112→   112→          return CHARS[Math.floor(Math.random() * CHARS.length)];
   113→   113→        }).join('')
   114→   114→      );
   115→   115→      iterRef.current += 0.5;
   116→   116→      if (iterRef.current < text.length) frameRef.current = requestAnimationFrame(scramble);
   117→   117→      else setDisplay(text);
   118→   118→    };
   119→   119→    frameRef.current = requestAnimationFrame(scramble);
   120→   120→    return () => cancelAnimationFrame(frameRef.current);
   121→   121→  }, [isHovered, text]);
   122→   122→
   123→   123→  return display;
   124→   124→}
