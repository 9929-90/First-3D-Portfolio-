    95→// ===== SCRAMBLE TEXT HOOK =====
    96→const CHARS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789@#$%';
    97→
    98→function useScramble(text, isHovered) {
    99→  const [display, setDisplay] = useState(text);
   100→  const frameRef = useRef(null);
   101→  const iterRef = useRef(0);
   102→
   103→  useEffect(() => {
   104→    if (!isHovered) { setDisplay(text); return; }
   105→    iterRef.current = 0;
   106→    const scramble = () => {
   107→      const iter = iterRef.current;
   108→      setDisplay(
   109→        text.split('').map((char, i) => {
   110→          if (char === ' ') return ' ';
   111→          if (i < iter) return text[i];
   112→          return CHARS[Math.floor(Math.random() * CHARS.length)];
   113→        }).join('')
   114→      );
   115→      iterRef.current += 0.5;
   116→      if (iterRef.current < text.length) frameRef.current = requestAnimationFrame(scramble);
   117→      else setDisplay(text);
   118→    };
   119→    frameRef.current = requestAnimationFrame(scramble);
   120→    return () => cancelAnimationFrame(frameRef.current);
   121→  }, [isHovered, text]);
   122→
   123→  return display;
   124→}

   539→const MagneticButton = ({ children, strength = 0.4 }) => {
   540→  const reducedMotion = useReducedMotion();
   541→  const x = useMotionValue(0); const y = useMotionValue(0);
   542→  const sx = useSpring(x, quickSpring); const sy = useSpring(y, quickSpring);

   544→    if (reducedMotion) return;
   545→    const { clientX, clientY, currentTarget } = e;
   546→    const { left, top, width, height } = currentTarget.getBoundingClientRect();
   547→    x.set(((clientX - left - width / 2) / width) * 20 * strength);
   548→    y.set(((clientY - top - height / 2) / height) * 20 * strength);
   549→  };

   552→      {children}
   553→    </motion.div>
